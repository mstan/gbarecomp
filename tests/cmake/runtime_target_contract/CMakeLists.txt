cmake_minimum_required(VERSION 3.20)
project(runtime_target_contract LANGUAGES C CXX)
include("${GBARECOMP_HELPER}")
set(GBARECOMP_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
foreach(lib gbarecomp_armv4t gbarecomp_gba gbarecomp_runtime gbarecomp_debug)
    add_library(${lib} INTERFACE)
endforeach()
set(gen "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${gen}")
file(WRITE "${gen}/recompiled_000.cpp" "int fixture_one() { return 1; }\n")
file(WRITE "${gen}/recompiled_001.cpp" "int fixture_two() { return 2; }\n")
file(WRITE "${gen}/dispatch_table.cpp" "int main() { return 0; }\n")
file(WRITE "${gen}/extra_support.c" "int fixture_extra(void) { return 3; }\n")
if(REJECT_MONOLITH)
    file(WRITE "${gen}/recompiled.cpp" "obsolete")
endif()
gbarecomp_collect_generated_sources(guest DIRECTORY "${gen}")
list(LENGTH guest count)
if(NOT count EQUAL 4)
    message(FATAL_ERROR "Generated support sources were lost")
endif()
add_executable(fixture ${guest})
gbarecomp_target_runtime(fixture)
gbarecomp_target_link_host_stack(fixture PRIVATE 268435456)
get_target_property(links fixture LINK_LIBRARIES)
get_target_property(options fixture LINK_OPTIONS)
if(TEST_FRONTEND)
    # Exercise ABI-specific configuration with the real target API; the normal
    # configure/build above still uses the host compiler and actual linker.
    set(CMAKE_EXECUTABLE_SUFFIX ".exe")
    set(CMAKE_CXX_COMPILER_ID "Clang")
    set(CMAKE_CXX_COMPILER_FRONTEND_VARIANT "${TEST_FRONTEND}")
    add_executable(abi_fixture ${guest})
    gbarecomp_target_runtime(abi_fixture)
    gbarecomp_target_link_host_stack(abi_fixture PRIVATE 67108864)
    get_target_property(links abi_fixture LINK_LIBRARIES)
    get_target_property(options abi_fixture LINK_OPTIONS)
    if(TEST_FRONTEND STREQUAL "MSVC")
        if(links MATCHES "start-group" OR NOT options MATCHES "/STACK:67108864")
            message(FATAL_ERROR "MSVC-style Clang received GNU linker flags")
        endif()
    elseif(NOT links MATCHES "start-group" OR NOT options MATCHES "--stack,67108864")
        message(FATAL_ERROR "GNU-style Clang lost its archive group or stack size")
    endif()
endif()
